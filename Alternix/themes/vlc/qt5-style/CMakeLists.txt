cmake_minimum_required(VERSION 3.10)
project(AlternixVLCStyle LANGUAGES CXX)
find_package(Qt5 5.12 REQUIRED COMPONENTS Widgets)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
add_library(alternixvlcstyle MODULE alternixvlcstyle.cpp)
target_link_libraries(alternixvlcstyle PRIVATE Qt5::Widgets)
set_target_properties(alternixvlcstyle PROPERTIES PREFIX "lib" OUTPUT_NAME "alternixvlcstyle")
